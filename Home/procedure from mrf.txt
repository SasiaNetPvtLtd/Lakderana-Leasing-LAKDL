  CREATE OR REPLACE PROCEDURE "MRFAC"."WEBACC_MAPLEDGER_ACC_FA" 
(
    M_PRODUCT_CODE     WEBAC_REF_FA_OTH_ACC.PORTFOLIO_CODE%TYPE, 
    M_ACC_TYPE_CODE    WEBAC_REF_FA_OTH_ACC.ACC_TYPE_CODE%TYPE,
    M_ACTYPE           WEBAC_REF_FA_OTH_ACC.ACTYPE%TYPE,
    M_ACODE1           WEBAC_REF_FA_OTH_ACC.ACODE1%TYPE,
    M_ACODE2           WEBAC_REF_FA_OTH_ACC.ACODE2%TYPE,
    M_ACODE3           WEBAC_REF_FA_OTH_ACC.ACODE3%TYPE,
    M_ACODE4           WEBAC_REF_FA_OTH_ACC.ACODE4%TYPE,
    M_LOCATION         WEBAC_REF_FA_OTH_ACC.LOCATION_CODE%TYPE --ADDED 2012-03-09 MADHAWA
)
AS
  M_COUNT              NUMBER(10);
  M_COUNT_NEW          NUMBER(10);

BEGIN
    
SELECT COUNT(*) INTO M_COUNT_NEW
FROM WEBAC_REF_FA_OTH_ACC
WHERE --PORTFOLIO_CODE = M_PORTFOLIO_CODE AND
     -- PORTFOLIO_CODE = M_PRODUCT_CODE AND 
      PRODUCT_CODE = M_PRODUCT_CODE AND
      ACC_TYPE_CODE = M_ACC_TYPE_CODE AND
      ACTYPE = M_ACTYPE AND
      ACODE1 = M_ACODE1 AND
      ACODE2 = M_ACODE2 AND
      ACODE3 = M_ACODE3 AND
      ACODE4 = M_ACODE4 AND
      LOCATION_CODE=M_LOCATION; --ADDED 2012-03-09 MADHAWA

IF M_COUNT_NEW>0 THEN
RAISE_APPLICATION_ERROR(-20201,'LEDGER ACCOUNT ALREADY MAPPED');        
END IF;

SELECT COUNT(PORTFOLIO_CODE) INTO M_COUNT
  FROM WEBAC_REF_FA_OTH_ACC
   WHERE --PORTFOLIO_CODE=M_PORTFOLIO_CODE AND
      PORTFOLIO_CODE = M_PRODUCT_CODE AND 
      ACC_TYPE_CODE=M_ACC_TYPE_CODE AND
      LOCATION_CODE=M_LOCATION; --ADDED 2012-03-09 MADHAWA

IF M_COUNT>0 THEN
       DELETE FROM WEBAC_REF_FA_OTH_ACC WHERE
         --PORTFOLIO_CODE=M_PORTFOLIO_CODE AND
         PORTFOLIO_CODE = M_PRODUCT_CODE AND 
         ACC_TYPE_CODE  = M_ACC_TYPE_CODE AND
         LOCATION_CODE  = M_LOCATION; --ADDED 2012-03-09 MADHAWA
END IF;

       INSERT INTO
             WEBAC_REF_FA_OTH_ACC(PORTFOLIO_CODE,ACC_TYPE_CODE,ACTYPE,ACODE1,ACODE2,ACODE3,ACODE4,LOCATION_CODE) --LOCATION_CODE ADD BY MADHAWA 2012-03-09
       VALUES
             (M_PRODUCT_CODE,M_ACC_TYPE_CODE,M_ACTYPE,M_ACODE1,M_ACODE2,M_ACODE3,M_ACODE4,M_LOCATION) ;
COMMIT;

END ;

    M_DIVISION_CODE    WEBAC_REF_DN_OTH_ACC.DIVISION_CODE%TYPE, --ADDED BY S.SHANHEETHA ON 17-07-2007
    M_PRODUCT_CODE     WEBAC_REF_DN_OTH_ACC.PRODUCT_CODE%TYPE,  --ADDED BY S.SHANHEETHA ON 17-07-2007
    M_ACC_TYPE_CODE    WEBAC_REF_DN_OTH_ACC.ACC_TYPE_CODE%TYPE,
    M_ACTYPE           WEBAC_REF_DN_OTH_ACC.ACTYPE %TYPE,
    M_ACODE1           WEBAC_REF_DN_OTH_ACC.ACODE1 %TYPE,
    M_ACODE2           WEBAC_REF_DN_OTH_ACC.ACODE2%TYPE,
    M_ACODE3           WEBAC_REF_DN_OTH_ACC.ACODE3%TYPE,
    M_ACODE4           WEBAC_REF_DN_OTH_ACC.ACODE4%TYPE,
    M_LOCATION_CODE    WEBAC_REF_USER_LOCATION.LOC_CODE%TYPE,
    M_NUM              INT            --ADDED BY PRABASH ON 27-03-2011
  )
AS
  M_COUNT              NUMBER(10);
  M_COUNT_NEW          NUMBER(10);
  M_REF_PRODUCT        WEBAC_REF_DN_OTH_ACC.ACODE4%TYPE;
  M_GEN_LOCATION       WEBAC_REF_USER_LOCATION.LOC_CODE%TYPE:='HO';
  M_REF_LOC            WEBAC_REF_DN_OTH_ACC.ACODE2%TYPE;

BEGIN

  IF  M_PRODUCT_CODE IS NOT NULL AND  M_LOCATION_CODE IS NOT NULL THEN
            IF M_NUM = 1 THEN
             INSERT INTO WEBAC_REF_DN_OTH_ACC_BK
             SELECT * FROM WEBAC_REF_DN_OTH_ACC WHERE

                     DIVISION_CODE = M_DIVISION_CODE AND    --ADDED BY S.SHANHEETHA ON 17-07-2007
                     PRODUCT_CODE = M_PRODUCT_CODE AND      --ADDED BY S.SHANHEETHA ON 17-07-2007
                     LOCATION_CODE = M_LOCATION_CODE   ;



             END IF;
                    DELETE FROM WEBAC_REF_DN_OTH_ACC WHERE
                     DIVISION_CODE = M_DIVISION_CODE AND    --ADDED BY S.SHANHEETHA ON 17-07-2007
                     PRODUCT_CODE = M_PRODUCT_CODE AND      --ADDED BY S.SHANHEETHA ON 17-07-2007
                     LOCATION_CODE = M_LOCATION_CODE AND         --ADDED BY PRABASH ON 27-03-2011
                     ACC_TYPE_CODE=M_ACC_TYPE_CODE;
                   IF  M_ACTYPE IS NOT NULL AND  M_ACODE1 IS NOT NULL AND M_ACODE2 IS NOT NULL AND M_ACODE3 IS NOT NULL AND M_ACODE4 IS NOT NULL THEN
                           INSERT INTO
                                 WEBAC_REF_DN_OTH_ACC(DIVISION_CODE,PRODUCT_CODE,ACC_TYPE_CODE,ACTYPE,ACODE1,ACODE2,ACODE3,ACODE4,LOCATION_CODE)
                           VALUES
                           (M_DIVISION_CODE,M_PRODUCT_CODE,M_ACC_TYPE_CODE,M_ACTYPE,M_ACODE1,M_ACODE2,M_ACODE3,M_ACODE4,M_LOCATION_CODE) ;
                   END IF;
   ELSIF  M_PRODUCT_CODE IS NOT NULL AND  M_LOCATION_CODE IS  NULL THEN

            FOR X IN (SELECT A.LOCATION_CODE, A.REF_LOCATION FROM WEBAC_REF_MAP_LOCATIONS A WHERE A.LOCATION_CODE<>M_GEN_LOCATION ) LOOP
              --raise_application_error(-20001,X.LOCATION_CODE);
               SELECT REF_PRODUCT INTO M_REF_PRODUCT FROM WEBAC_REF_MAP_PRODUCT WHERE REF_SYSTEM='LEASING' AND PRODUCT_CODE=M_PRODUCT_CODE;
               SELECT REF_LOCATION INTO M_REF_LOC FROM WEBAC_REF_MAP_LOCATIONS WHERE LOCATION_CODE=M_GEN_LOCATION;


               FOR Y IN (SELECT A.ACTYPE, A.ACODE1, X.REF_LOCATION ACODE2, A.ACODE3, A.ACODE4, WEBACC_GET_ACT_DESC(A.ACTYPE, A.ACODE1, X.REF_LOCATION, A.ACODE3, A.ACODE4) ACDESC,
                        A.ACT_STATUS, A.HEAD_CODE, A.SHEAD_CODE, A.DISPLAY_STATUS,A.ENT_DATE,A.ENT_USER,
                         A.MOD_DATE,A.MOD_USER
                   FROM WEBAC_REF_ACCOUNTS A WHERE A.ACODE4=M_REF_PRODUCT AND A.ACODE2=M_REF_LOC ) LOOP

                    SELECT COUNT(*) INTO M_COUNT FROM WEBAC_REF_ACCOUNTS A
                    WHERE A.ACTYPE = Y.ACTYPE AND
                          A.ACODE1 = Y.ACODE1 AND
                          A.ACODE2 = Y.ACODE2 AND
                          A.ACODE3 = Y.ACODE3 AND
                          A.ACODE4 = Y.ACODE4;
                    IF M_COUNT = 0 THEN
                       INSERT INTO WEBAC_REF_ACCOUNTS VALUES (
                         Y.ACTYPE, Y.ACODE1, Y.ACODE2, Y.ACODE3, Y.ACODE4,Y.ACDESC,Y.ACT_STATUS, Y.HEAD_CODE, Y.SHEAD_CODE, Y.DISPLAY_STATUS
                       ,Y.ENT_DATE,Y.ENT_USER,Y.MOD_DATE,Y.MOD_USER
                       );


                    END IF;


               END LOOP;

               --INSERT INTO WEBAC_REF_DN_OTH_ACC

                  FOR K IN
                       (SELECT  A.DIVISION_CODE, A.PRODUCT_CODE, A.ACC_TYPE_CODE, A.ACTYPE,
                       A.ACODE1, X.REF_LOCATION ACODE2, A.ACODE3, A.ACODE4, X.LOCATION_CODE LOCATION_CODE FROM WEBAC_REF_DN_OTH_ACC A
                       WHERE A.PRODUCT_CODE = M_PRODUCT_CODE AND A.LOCATION_CODE = M_GEN_LOCATION ) LOOP


                       /*SELECT COUNT(*) INTO M_COUNT FROM WEBAC_REF_DN_OTH_ACC A
                        WHERE A.ACTYPE = K.ACTYPE AND
                          A.ACODE1 = K.ACODE1 AND
                          A.ACODE2 = K.ACODE2 AND
                          A.ACODE3 = K.ACODE3 AND
                          A.ACODE4 = K.ACODE4 AND
                          A.DIVISION_CODE = K.DIVISION_CODE AND
                          A.PRODUCT_CODE = K.PRODUCT_CODE AND
                          A.ACC_TYPE_CODE = K.ACC_TYPE_CODE AND
                          A.LOCATION_CODE = K.LOCATION_CODE;*/
                          
                            SELECT COUNT(*) INTO M_COUNT FROM WEBAC_REF_DN_OTH_ACC A
                        WHERE 
                          A.DIVISION_CODE = K.DIVISION_CODE AND
                          A.PRODUCT_CODE = K.PRODUCT_CODE AND
                          A.ACC_TYPE_CODE = K.ACC_TYPE_CODE AND
                          A.LOCATION_CODE = K.LOCATION_CODE;

                   IF M_COUNT = 0 THEN

                    SELECT COUNT(*) INTO M_COUNT FROM WEBAC_REF_ACCOUNTS A
                      WHERE A.ACTYPE = K.ACTYPE AND
                          A.ACODE1 = K.ACODE1 AND
                          A.ACODE2 = K.ACODE2 AND
                          A.ACODE3 = K.ACODE3 AND
                          A.ACODE4 = K.ACODE4;

                          IF M_COUNT = 1 THEN
                             INSERT INTO WEBAC_REF_DN_OTH_ACC VALUES
                      (K.DIVISION_CODE,K.PRODUCT_CODE,K.ACC_TYPE_CODE,K.ACTYPE,K.ACODE1,K.ACODE2,K.ACODE3,K.ACODE4,K.LOCATION_CODE);
                            ELSE
                         --   RAISE_APPLICATION_ERROR(-20001,'INVALID ACCOUTNS MAPPING FOR LOCATION-' || K.DIVISION_CODE || ' ' || K.PRODUCT_CODE|| ' ' ||K.ACC_TYPE_CODE|| ' ' ||K.ACTYPE|| ' ' ||K.ACODE1|| ' ' ||K.ACODE2|| ' ' ||K.ACODE3|| ' ' ||K.ACODE4|| ' ' ||K.LOCATION_CODE);
                         NULL;
                          END IF;




                   END IF;

                   END LOOP;
     END LOOP;





   END IF;
   COMMIT;

END WEBACC_MAPLEDGER_ACC;
